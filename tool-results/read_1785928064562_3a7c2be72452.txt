   119→export interface CliProbeResult {
   120→  id: string;
   121→  label: string;
   122→  category: string;
   123→  available: boolean;
   124→  version?: string;
   125→  bin: string;
   126→  description: string;
   127→  capabilities: string[];
   128→  error?: string;
   129→}
   130→
   131→// In-process cache so we don't re-probe on every request.
   132→let cache: CliProbeResult[] | null = null;
   133→let cacheTime = 0;
   134→const CACHE_TTL_MS = 60_000; // 1 min
   135→
   136→async function probeOne(adapter: CliAdapter): Promise<CliProbeResult> {
   137→  const base: CliProbeResult = {
   138→    id: adapter.id,
   139→    label: adapter.label,
   140→    category: adapter.category,
   141→    available: false,
   142→    bin: adapter.bin,
   143→    description: adapter.description,
   144→    capabilities: adapter.capabilities,
   145→  };
   146→
   147→  try {
   148→    if (adapter.category === "binary") {
   149→      try {
   150→        await execFileAsync("which", [adapter.bin], { timeout: 3000 });
   151→      } catch {
   152→        return { ...base, error: "binary not in PATH" };
   153→      }
   154→      if (adapter.probeArgs.length > 0) {
   155→        try {
   156→          const { stdout } = await execFileAsync(adapter.bin, adapter.probeArgs, {
   157→            timeout: 5000,
   158→            maxBuffer: 1024 * 64,
   159→          });
   160→          const version = stdout.trim().split("\n")[0].slice(0, 80);
   161→          return { ...base, available: true, version };
   162→        } catch {
   163→          return { ...base, available: true, version: "ok" };
   164→        }
   165→      }
   166→      return { ...base, available: true };
   167→    }
   168→
   169→    if (adapter.category === "python") {
   170→      const { stdout } = await execFileAsync(
   171→        adapter.bin,
   172→        adapter.probeArgs,
   173→        { timeout: 5000, maxBuffer: 1024 * 64 }
   174→      );
   175→      const version = stdout.trim().split("\n")[0].slice(0, 80);
   176→      return { ...base, available: true, version };
   177→    }
   178→
   179→    if (adapter.category === "pymol") {
   180→      try {
   181→        await execFileAsync("which", [adapter.bin], { timeout: 3000 });
   182→        return { ...base, available: true, version: "ok" };
   183→      } catch {
   184→        return { ...base, error: "pymol not in PATH" };
   185→      }
   186→    }
   187→
   188→    return base;
   189→  } catch (err) {
   190→    return {
   191→      ...base,
   192→      error: err instanceof Error ? err.message : String(err),
   193→    };
   194→  }
   195→}
   196→
   197→export async function probeAllClis(force = false): Promise<CliProbeResult[]> {
   198→  if (cache && !force && Date.now() - cacheTime < CACHE_TTL_MS) {
   199→    return cache;
   200→  }
   201→  const results = await Promise.all(CLI_ADAPTERS.map(probeOne));
   202→  cache = results;
   203→  cacheTime = Date.now();
   204→  return results;
   205→}
