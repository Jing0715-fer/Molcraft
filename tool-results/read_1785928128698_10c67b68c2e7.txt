    30→// ===== Types for each analysis result =====
    31→interface SummaryData {
    32→  chain_count: number;
    33→  chains: Record<string, { residue_count: number; first_resno: number | null; last_resno: number | null; atom_count: number }>;
    34→  total_residues: number;
    35→  total_atoms: number;
    36→  ligands: Record<string, number>;
    37→  has_hydrogens: boolean;
    38→}
    39→
    40→interface RamaData {
    41→  total_residues: number;
    42→  favoured_pct: number;
    43→  outlier_pct: number;
    44→  regions: Record<string, number>;
    45→}
    46→
    47→interface SsData {
    48→  total_residues: number;
    49→  ss_counts: { alpha_helix: number; beta_sheet: number; coil: number; turn: number };
    50→  alpha_helix_pct: number;
    51→  beta_sheet_pct: number;
    52→  coil_pct: number;
    53→  turn_pct: number;
    54→}
    55→
    56→interface BfactorData {
    57→  total_chains: number;
    58→  chains: Record<string, { mean: number; min: number; max: number; std: number; is_plddt: boolean }>;
    59→}
    60→
    61→interface SasaData {
    62→  total_sasa_A2: number;
    63→  chain_sasa_A2: Record<string, number>;
    64→  n_chains: number;
    65→}
    66→
    67→interface DisulfideData {
    68→  count: number;
    69→  bonds: Array<{ chain1: string; resno1: number; chain2: string; resno2: number; distance_A: number }>;
    70→  cutoff: number;
    71→}
    72→
    73→interface OligomerData {
    74→  n_chains: number;
    75→  oligomer_type: string;
    76→  is_homomer: boolean;
    77→  n_interfaces: number;
    78→  chains: Array<{ chain: string; residue_count: number; atom_count: number }>;
    79→  interfaces: Array<{ chain1: string; chain2: string; contact_atoms: number; min_distance_A: number }>;
    80→}
    81→
    82→interface ValidationData {
    83→  quality: "good" | "fair" | "poor";
    84→  clash_count: number;
    85→  rama_outlier_count: number;
    86→  rama_outlier_pct: number;
    87→  missing_sidechain_count: number;
    88→}
    89→
    90→interface OverviewState {
    91→  summary: SummaryData | null;
    92→  rama: RamaData | null;
    93→  ss: SsData | null;
    94→  bfactor: BfactorData | null;
    95→  sasa: SasaData | null;
    96→  disulfide: DisulfideData | null;
    97→  oligomer: OligomerData | null;
    98→  validation: ValidationData | null;
    99→  errors: string[];
   100→  loaded: boolean;
   101→}
   102→

   104→  summary: null,

   109→  disulfide: null,
   110→  oligomer: null,
   111→  validation: null,
   112→  errors: [],
   113→  loaded: false,
   114→};

   118→  fair: { label: "一般", color: "#f59e0b", bg: "#f59e0b20" },

   135→
   136→export function StructureOverviewDashboard() {
   137→  const activeStructure = useAppStore(selectActiveStructure);
   138→  const structureFileCache = useAppStore((s) => s.structureFileCache);
   139→  const toast = useAppStore((s) => s.toast);
   140→  const viewer = useAppStore((s) => s.viewer);
   141→  const [state, setState] = useState<OverviewState>(INITIAL_STATE);
   142→  const [loading, setLoading] = useState(false);
   143→  const [progress, setProgress] = useState(0);
   144→  const progressRef = useRef(0);
   145→
   146→  const activeId = activeStructure?.id;
   147→  const isPdbId = activeId ? /^[a-zA-Z0-9]{4}$/.test(activeId) : false;
   148→  const hasFileCache = activeId ? !!structureFileCache[activeId] : false;
   149→
   150→  const fetchAll = useCallback(async () => {
   151→    if (!activeId) {
   152→      setState(INITIAL_STATE);
   153→      return;
   154→    }
   155→
   156→    const body: Record<string, unknown> = {};
   157→    if (isPdbId) {
   158→      body.pdbId = activeId;
   159→    } else if (hasFileCache) {
   160→      body.fileContent = structureFileCache[activeId].content;
   161→      body.fileFormat = structureFileCache[activeId].format;
   162→    } else {
   163→      setState({
   164→        ...INITIAL_STATE,
   165→        errors: [`当前结构 (${activeId}) 不是 PDB ID 且无本地文件缓存，无法生成概览。`],
   166→      });
   167→      return;
   168→    }
   169→
   170→    setLoading(true);
   171→    setProgress(0);
   172→    progressRef.current = 0;
   173→    setState({ ...INITIAL_STATE, loaded: false });
   174→
   175→    const recipes = [
   176→      { key: "summary" as const, recipe: "summary" },
   177→      { key: "rama" as const, recipe: "ramachandran" },
   178→      { key: "ss" as const, recipe: "secondary_structure_simple" },
   179→      { key: "bfactor" as const, recipe: "bfactor_stats" },
   180→      { key: "sasa" as const, recipe: "sasa" },
   181→      { key: "disulfide" as const, recipe: "disulfide_bonds" },
   182→      { key: "oligomer" as const, recipe: "oligomer_analysis" },
   183→      { key: "validation" as const, recipe: "structure_validation" },
   184→    ];
   185→
   186→    const errors: string[] = [];
   187→    const results: Partial<OverviewState> = {};
   188→
   189→    // Run all recipes in parallel for speed
   190→    await Promise.all(
   191→      recipes.map(async ({ key, recipe }) => {
   192→        try {
   193→          const res = await fetch("/api/analyze/run", {
   194→            method: "POST",
   195→            headers: { "Content-Type": "application/json" },
   196→            body: JSON.stringify({ ...body, recipe }),
   197→          });
   198→          if (!res.ok) {
   199→            const err = await res.json().catch(() => ({}));
   200→            throw new Error(err.detail || err.error || `HTTP ${res.status}`);
   201→          }
   202→          const json = await res.json();
   203→          if (json.data) {
   204→            if (json.data.error) {
   205→              errors.push(`${recipe}: ${json.data.error}`);
   206→            } else {
   207→              results[key] = json.data;
   208→            }
   209→          } else {
   210→            errors.push(`${recipe}: 无数据`);
   211→          }
   212→        } catch (err) {
   213→          const msg = err instanceof Error ? err.message : String(err);
   214→          errors.push(`${recipe}: ${msg}`);
   215→        }
   216→        progressRef.current += 1;
   217→        setProgress(Math.round((progressRef.current / recipes.length) * 100));
   218→      })
   219→    );
   220→
   221→    setState({
   222→      ...(results as OverviewState),
   223→      errors,
   224→      loaded: true,
   225→    });
   226→    setLoading(false);
   227→    if (errors.length > 0 && errors.length < recipes.length) {
   228→      toast(`概览完成 (${recipes.length - errors.length}/${recipes.length} 成功)`, "info");
   229→    } else if (errors.length === 0) {
   230→      toast("概览分析完成", "success");
   231→    } else {
   232→      toast(`概览失败: ${errors.length} 项错误`, "error");
   233→    }
   234→  }, [activeId, isPdbId, hasFileCache, structureFileCache, toast]);
   235→
   236→  useEffect(() => {
   237→    fetchAll();
   238→  }, [fetchAll]);
   239→
   240→  /** Export all analysis results as a markdown report */
   241→  const handleExport = useCallback(() => {
   242→    if (!state.loaded || !activeId) return;
   243→    const label = activeStructure?.label ?? activeId;
   244→    const lines: string[] = [];
   245→    lines.push(`# 结构概览报告: ${label}`);
   246→    lines.push("");
   247→    lines.push(`> 生成时间: ${new Date().toLocaleString("zh-CN")}`);
   248→    lines.push(`> 分析项: 8 项并行 (摘要 / Ramachandran / 二级结构 / B-factor / SASA / 二硫键 / 寡聚体 / 验证)`);
   249→    lines.push("");
   250→
   251→    // Summary
   252→    if (state.summary) {
   253→      lines.push("## 1. 结构摘要");
   254→      lines.push("");
   255→      lines.push(`- **链数**: ${state.summary.chain_count}`);
   256→      lines.push(`- **总残基数**: ${state.summary.total_residues}`);
   257→      lines.push(`- **总原子数**: ${state.summary.total_atoms}`);
   258→      lines.push(`- **含氢原子**: ${state.summary.has_hydrogens ? "是" : "否"}`);
   259→      if (Object.keys(state.summary.ligands).length > 0) {
   260→        lines.push(`- **配体**: ${Object.entries(state.summary.ligands).map(([k, v]) => `${k}${v > 1 ? `×${v}` : ""}`).join(", ")}`);
   261→      }
   262→      lines.push("");
   263→      lines.push("### 链详情");
   264→      lines.push("");
   265→      lines.push("| 链 | 残基数 | 原子数 | 范围 |");
   266→      lines.push("|---|---|---|---|");
   267→      for (const [chain, info] of Object.entries(state.summary.chains)) {
   268→        lines.push(`| ${chain} | ${info.residue_count} | ${info.atom_count} | ${info.first_resno}-${info.last_resno} |`);
   269→      }
   270→      lines.push("");
   271→    }
   272→
   273→    // Validation
   274→    if (state.validation) {
   275→      lines.push("## 2. 结构质量验证");
   276→      lines.push("");
   277→      lines.push(`- **质量评分**: ${QUALITY_META[state.validation.quality].label}`);
   278→      lines.push(`- **原子碰撞数**: ${state.validation.clash_count}`);
   279→      lines.push(`- **Ramachandran 异常率**: ${state.validation.rama_outlier_pct}% (${state.validation.rama_outlier_count}/${state.validation.total_phi_psi})`);
   280→      lines.push(`- **缺失侧链数**: ${state.validation.missing_sidechain_count}`);
   281→      lines.push("");
   282→    }
   283→
   284→    // Ramachandran
   285→    if (state.rama) {
   286→      lines.push("## 3. Ramachandran 分析");
   287→      lines.push("");
   288→      lines.push(`- **总残基数**: ${state.rama.total_residues}`);
   289→      lines.push(`- **偏好区**: ${state.rama.favoured_pct}%`);
   290→      lines.push(`- **异常区**: ${state.rama.outlier_pct}%`);
   291→      lines.push("");
   292→    }
   293→
   294→    // Secondary structure
   295→    if (state.ss) {
   296→      lines.push("## 4. 二级结构组成");
   297→      lines.push("");
   298→      lines.push(`- **总残基数**: ${state.ss.total_residues}`);
   299→      lines.push(`- **α-螺旋**: ${state.ss.alpha_helix_pct}% (${state.ss.ss_counts.alpha_helix} 残基)`);
   300→      lines.push(`- **β-折叠**: ${state.ss.beta_sheet_pct}% (${state.ss.ss_counts.beta_sheet} 残基)`);
   301→      lines.push(`- **转角**: ${state.ss.turn_pct}% (${state.ss.ss_counts.turn} 残基)`);
   302→      lines.push(`- **无规卷曲**: ${state.ss.coil_pct}% (${state.ss.ss_counts.coil} 残基)`);
   303→      lines.push("");
   304→    }
   305→
   306→    // B-factor
   307→    if (state.bfactor) {
   308→      lines.push("## 5. B-factor / pLDDT 统计");
   309→      lines.push("");
   310→      lines.push("| 链 | 均值 | 最小 | 最大 | 标准差 | 类型 |");
   311→      lines.push("|---|---|---|---|---|---|");
   312→      for (const [chain, info] of Object.entries(state.bfactor.chains)) {
   313→        lines.push(`| ${chain} | ${info.mean} | ${info.min} | ${info.max} | ${info.std} | ${info.is_plddt ? "pLDDT" : "B-factor"} |`);
   314→      }
   315→      lines.push("");
   316→    }
   317→
   318→    // SASA
   319→    if (state.sasa) {
   320→      lines.push("## 6. 溶剂可及表面积 (SASA)");
   321→      lines.push("");
   322→      lines.push(`- **总 SASA**: ${state.sasa.total_sasa_A2.toFixed(2)} Å²`);
   323→      lines.push(`- **链数**: ${state.sasa.n_chains}`);
   324→      lines.push("");
   325→      lines.push("| 链 | SASA (Å²) | 占比 |");
   326→      lines.push("|---|---|---|");
   327→      for (const [chain, sasa] of Object.entries(state.sasa.chain_sasa_A2)) {
   328→        const pct = (sasa / state.sasa.total_sasa_A2) * 100;
   329→        lines.push(`| ${chain} | ${sasa.toFixed(2)} | ${pct.toFixed(1)}% |`);
   330→      }
   331→      lines.push("");
   332→    }
   333→
   334→    // Disulfide bonds
   335→    if (state.disulfide) {
   336→      lines.push("## 7. 二硫键");
   337→      lines.push("");
   338→      lines.push(`- **检测数**: ${state.disulfide.count} (截断 ${state.disulfide.cutoff} Å)`);
   339→      if (state.disulfide.bonds.length > 0) {
   340→        lines.push("");
   341→        lines.push("| 残基 1 | 残基 2 | 距离 (Å) |");
   342→        lines.push("|---|---|---|");
   343→        for (const bond of state.disulfide.bonds) {
   344→          lines.push(`| CYS${bond.resno1} (${bond.chain1}) | CYS${bond.resno2} (${bond.chain2}) | ${bond.distance_A.toFixed(3)} |`);
   345→        }
   346→      }
   347→      lines.push("");
   348→    }
   349→
   350→    // Oligomer
   351→    if (state.oligomer) {
   352→      lines.push("## 8. 寡聚体分析");
   353→      lines.push("");
   354→      lines.push(`- **寡聚类型**: ${state.oligomer.oligomer_type}`);
   355→      lines.push(`- **链数**: ${state.oligomer.n_chains}`);
   356→      lines.push(`- **界面数**: ${state.oligomer.n_interfaces}`);
   357→      lines.push(`- **对称性**: ${state.oligomer.is_homomer ? "同源寡聚体" : "异源寡聚体"}`);
   358→      if (state.oligomer.interfaces.length > 0) {
   359→        lines.push("");
   360→        lines.push("### 链间界面");
   361→        lines.push("");
   362→        lines.push("| 链 1 | 链 2 | 接触原子数 | 最小距离 (Å) |");
   363→        lines.push("|---|---|---|---|");
   364→        for (const iface of state.oligomer.interfaces) {
   365→          lines.push(`| ${iface.chain1} | ${iface.chain2} | ${iface.contact_atoms} | ${iface.min_distance_A.toFixed(2)} |`);
   366→        }
   367→      }
   368→      lines.push("");
   369→    }
   370→
   371→    // Errors
   372→    if (state.errors.length > 0) {
   373→      lines.push("## ⚠️ 失败的分析");
   374→      lines.push("");
   375→      for (const err of state.errors) {
   376→        lines.push(`- ${err}`);
   377→      }
   378→      lines.push("");
   379→    }
   380→
   381→    lines.push("---");
   382→    lines.push("");
   383→    lines.push("*报告由 MolCraft AI 自动生成 · 基于 Molstar + Biopython + FreeSASA*");
   384→
   385→    const md = lines.join("\n");
   386→    const blob = new Blob([md], { type: "text/markdown;charset=utf-8" });
   387→    const url = URL.createObjectURL(blob);
   388→    const a = document.createElement("a");
   389→    a.href = url;
   390→    a.download = `overview-${label.replace(/[^\w\u4e00-\u9fa5-]+/g, "_")}.md`;
   391→    a.click();
   392→    URL.revokeObjectURL(url);
   393→    toast("概览报告 (Markdown) 已导出", "success");
   394→  }, [state, activeId, activeStructure, toast]);
   395→
   396→  /** Export all analysis results as a styled HTML report */
   397→  const handleExportHtml = useCallback(() => {
   398→    if (!state.loaded || !activeId) return;
   399→    const label = activeStructure?.label ?? activeId;
   400→    const date = new Date().toLocaleString("zh-CN");
   401→    const qualityColor = state.validation ? QUALITY_META[state.validation.quality].color : "#6b7280";
   402→    const qualityLabel = state.validation ? QUALITY_META[state.validation.quality].label : "—";
   403→
   404→    const card = (title: string, content: string, bg = "#f8fafc") =>
   405→      `<div style="background:${bg};border:1px solid #e2e8f0;border-radius:8px;padding:12px;margin-bottom:12px"><h3 style="margin:0 0 8px;color:#0f766e;font-size:13px;border-bottom:1px solid #e2e8f0;padding-bottom:4px">${title}</h3>${content}</div>`;
   406→
   407→    const metric = (label: string, value: string, color: string) =>
   408→      `<div style="text-align:center;padding:8px;border:1px solid #e2e8f0;border-radius:6px;background:#fff"><div style="font-size:9px;color:#64748b;text-transform:uppercase">${label}</div><div style="font-family:monospace;font-weight:bold;font-size:14px;color:${color}">${value}</div></div>`;
   409→
   410→    let html = `<!DOCTYPE html><html lang="zh-CN"><head><meta charset="UTF-8"><meta name="viewport" content="width=device-width,initial-scale=1"><title>结构概览报告: ${label}</title><style>body{font-family:-apple-system,BlinkMacSystemFont,"Segoe UI",sans-serif;max-width:900px;margin:20px auto;padding:0 16px;color:#1e293b;line-height:1.5}h1{color:#0f766e;border-bottom:2px solid #0f766e;padding-bottom:6px}h2{color:#0369a1;margin-top:20px}table{border-collapse:collapse;width:100%;font-size:12px}th,td{border:1px solid #cbd5e1;padding:4px 8px;text-align:left}th{background:#f1f5f9}.meta{color:#64748b;font-size:11px}</style></head><body>`;
   411→    html += `<h1>🔬 结构概览报告: ${label}</h1><p class="meta">生成时间: ${date} · MolCraft AI · 8 项并行分析</p>`;
   412→
   413→    // Identity + Quality
   414→    html += '<div style="display:grid;grid-template-columns:1fr 1fr;gap:12px;margin-bottom:12px">';
   415→    if (state.summary) {
   416→      html += card("结构标识", `<div style="font-family:monospace;font-size:16px;font-weight:bold">${label}</div><div style="font-size:11px;color:#64748b">${state.summary.chain_count} 链 · ${state.summary.total_residues} 残基 · ${state.summary.total_atoms} 原子</div>`, "#ecfdf5");
   417→    }
   418→    if (state.validation) {
   419→      html += card("质量评分", `<div style="font-size:16px;font-weight:bold;color:${qualityColor}">${qualityLabel}</div><div style="font-size:11px;color:#64748b">碰撞 ${state.validation.clash_count} · 拉氏异常 ${state.validation.rama_outlier_pct}% · 缺侧链 ${state.validation.missing_sidechain_count}</div>`, qualityColor + "15");
   420→    }
   421→    html += '</div>';
   422→
   423→    // Key metrics
   424→    html += '<div style="display:grid;grid-template-columns:repeat(4,1fr);gap:8px;margin-bottom:12px">';
   425→    if (state.rama) html += metric("偏好区", `${state.rama.favoured_pct}%`, "#10b981");
   426→    if (state.ss) html += metric("α/β 比", `${state.ss.alpha_helix_pct}/${state.ss.beta_sheet_pct}`, "#3b82f6");
   427→    if (state.bfactor) {
   428→      const first = Object.values(state.bfactor.chains)[0];
   429→      html += metric("B 均值", first ? first.mean.toFixed(1) : "—", "#8b5cf6");
   430→    }
   431→    if (state.sasa) html += metric("总 SASA", `${(state.sasa.total_sasa_A2 / 1000).toFixed(1)}k`, "#06b6d4");
   432→    html += '</div>';
   433→
   434→    // Summary
   435→    if (state.summary) {
   436→      let table = '<table><thead><tr><th>链</th><th>残基</th><th>原子</th><th>范围</th></tr></thead><tbody>';
   437→      for (const [chain, info] of Object.entries(state.summary.chains)) {
   438→        table += `<tr><td><b>${chain}</b></td><td>${info.residue_count}</td><td>${info.atom_count}</td><td>${info.first_resno}-${info.last_resno}</td></tr>`;
   439→      }
